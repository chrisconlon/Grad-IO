\section*{Nonparametrics?}


\begin{frame}{What do you mean by non-parametric?}
Mostly we mean putting a flexible distribution on $f(\beta_i,\alpha_i \mid \theta)$ (and keeping logit error on $\varepsilon_{ij}$)
\begin{align*}
u_{ij} = \beta_i x_j - \alpha_i p_j + \xi_j + \varepsilon_{ij} \text{ with } f(\beta_i,\alpha_i \mid \theta)
\end{align*}
\begin{itemize}
    \item Fixed Grids (Fox, Kim, Ryan, Bajari 2011, Heiss, Hetzenecker, Osterhaus 2022, Nevo Turner Williams 2016): draw from ``prior'' of $\beta_i$, compute $\sigma_{ij}(\beta_i)$ and choose weights on each $i$, $w_i$.
    \item Compiani (QE 2022): approximate $\sigma_{j}^{-1}(\symbf{s}_t,\symbf{x}_t^{(2)})$ directly with Bernstein Polynomials (ditches $\varepsilon \rightarrow$ very hard)
    \item Ao Wang (JE 2022): use polynomial sieves: $\mathbb{E}\left[\left(\sigma_j^{-1}\left(\symbf{s}_t ; \symbf{x}_t^{(2)}, \alert{F}\right)-X_t^{(1)} \beta^{(1)}\right) \phi_k\left(Z_{j t}\right)\right]=0$
    \item Lu, Shi, Tao (JE 2023): use partially linear model: $\log \left(s_{j t} / s_{0 t}\right)=X_{1, j t}^{\prime} \beta^0+\alert{\psi^0\left(X_{2, j t} ; IV_{J, t}\right)}+\xi_{j t}$ where        $\psi^0\left(x_{2, j t} ; IV_{J, t}\right)=\log \left[\frac{\int \frac{\exp \left(x_{2, j t}^{\prime} v\right)}{\exp \left(IV_{J, t}(v)\right)} f^0(v) d v}{\int \frac{1}{\exp \left(IV_{J, t}(v)\right)} f^0(v) d v}\right]$.
\end{itemize}
But these are still \alert{only as good as characteristics}.
\end{frame}

\begin{frame}{Idea \#1: t-STE Embeddings (Van Der Maaten and Weinberger, 2012)}
\begin{itemize}
  \item \textbf{Goal:} Assign a low-dimensional vector of characteristics $x_{j} \in \mathbb{R}^m$ for each product using triplets of the form
  “$j$ is more similar to $k$ than to $\ell$”.
  \vspace{0.6em}
  \item The “t” in t-STE comes from using a \textbf{Student-t kernel} 
  to model distances—giving heavy tails and better separation of clusters.
  \begin{align*}
\max _{\mathbf{x}} \sum_{(j, k,\ell) \in \mathcal{T}} \ln \left(\pi_{j k \ell}\right) \quad \text { where } \quad \pi_{ j k \ell}=\frac{\left(1+\frac{\left\|x_j-x_{\ell}\right\|^2}{\alpha}\right)^{-\frac{\alpha+1}{2}}}{\left(1+\frac{\left\|x_j-x_k\right\|^2}{\alpha}\right)^{-\frac{\alpha+1}{2}}+\left(1+\frac{\left\|x_j-x_{\ell}\right\|^2}{\alpha}\right)^{-\frac{\alpha+1}{2}}}
  \end{align*}
  \item This is basically maximium (log) likelihood for a binary outcome model (closer to $\ell$ than $k$).
  \begin{itemize}
        \item Fit with canned routine (which does gradient descent).
        \item $\alpha$ is a (somewhat arbitrary) tuning parameter.
   \end{itemize}

\end{itemize}
\end{frame}

\begin{frame}{Second Choices as a Matrix: How Much Rank Do You Need?}
For any mixture of logits, the second-choice (product-removal) diversion from $j$ to $k$ is
\begin{align*}
D_{j \rightarrow k} = \sum_{i=1}^I w_i \cdot \frac{\sigma_{ij}}{1-\sigma_{ij}} \cdot \frac{\sigma_{ik}}{s_j}
\quad\Longleftrightarrow\quad
\symbf{D} = \text{diag}(\symbf{s})^{-1}\, \widetilde{\symbf{S}}\, \text{diag}(w)\, \symbf{S}^{\top},
\qquad \widetilde{\sigma}_{ij} = \tfrac{\sigma_{ij}}{1-\sigma_{ij}},
\end{align*}
where the columns of $\symbf{S}$ are the type-level choice-probability vectors $\symbf{\sigma}_i$.
\vspace{-0.1cm}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\small
\begin{tabular}{lcl}
\toprule
\textbf{Model} & \textbf{rank}$(\symbf{D})$ & \textbf{Why} \\
\midrule
Logit ($I=1$) & $\leq 1$ & one outer product $\widetilde{\symbf{s}}\,\symbf{s}^{\top}$: proportional substitution \\
Nested logit ($G$ nests) & $\leq G$ & one outer product per (non-singleton) nest \\
Mixed logit ($I$ types) & $\leq \min\{I, J+1\}$ & one rank-one term per type \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.1cm}
\begin{wideitemize}
    \item Logit and nested-logit bounds are \alert{tight}; the mixed-logit bound is usually slack (thousands of draws on a handful of characteristics).
    \item Idea: skip characteristics and parametric random coefficients; \alert{control $\text{rank}(\symbf{D})$ directly} in product space.
\end{wideitemize}
\end{frame}


\begin{frame}{Conlon, Mortimer, Sarkis (2026): Second Choices in Product Space}
\small
Data from a \alert{single market}: shares $s_j$ and a subset $\Omega$ of survey second-choice cells $\calD_{j \rightarrow k}$. For each rank $I$, solve
\begin{align*}
\min_{(\symbf{S},\, w) \geq 0}\
\sum_{(j,k) \in \Omega} \omega_{jk}\, \bigl(\calD_{j \rightarrow k} - D_{j \rightarrow k}(\symbf{S},w)\bigr)^2
\;+\; \lambda\, \bigl\| \symbf{s} - \symbf{S}\, w \bigr\|_{\ell_2}^2
\quad \text{s.t.}\quad
\textstyle\sum_j \sigma_{ij} = 1\ \forall i,\quad \sum_i w_i = 1.
\end{align*}
\vspace{-0.4cm}
\begin{wideitemize}
\item \textbf{Estimates} $\symbf{\sigma}_i$ (type profiles) and $w_i$ (type weights): a finite mixture with $I \ll J$ and no characteristics.
\item \textbf{Tuning}: $\lambda$ is a share-fit penalty (not a ridge); $(I,\lambda)$ chosen by cross-validation on held-out cells of $\Omega$.
\item \textbf{Weights} $\omega_{jk} \propto n_j$, the respondents behind row $j$ (approximately inverse-variance under row-multinomial sampling).
\item \textbf{Computation}: reparameterize $\sigma_{ij} = e^{V_{ij}}/\sum_k e^{V_{ik}}$, $w_i = e^{\gamma_i}/\sum_{\iota} e^{\gamma_\iota}$, so the problem is unconstrained; autodiff + Adam. Optional sparsification zeroes $\geq 75\%$ of $\symbf{S}$ (stop when AIC rises).
\item \textbf{Matrix completion}: the fitted $\symbf{D}(\symbf{S},w)$ predicts the \emph{unobserved} cells $(j,k) \notin \Omega$.
\item Software: \href{https://github.com/metalforpc/PyCMS}{\texttt{PyCMS}} (PyTorch).
\end{wideitemize}
\end{frame}

\begin{frame}{Second Stage: Putting Prices Back In}
\small
The first stage delivers $\widehat{\symbf{S}}, \widehat{w}$ and hence $V_{ij} - V_{i0}$, but no price sensitivity and no welfare. Assume $V_{ij} = V^{*}_{ij} - \alpha_i p_j$ (constant within type, free across types) and calibrate $\alpha_i > 0$ from one of:
\vspace{-0.15cm}
\begin{wideitemize}
\item \textbf{Own-price elasticities} (a quasi-experiment, another study, a simpler demand system):
\begin{align*}
\min_{\alpha > 0}\ \Bigl\| \mathcal{E}_{jj} + \frac{p_j}{\widehat{s}_j} \sum_{i=1}^I \alpha_i\, \widehat{w}_i\, \widehat{\sigma}_{ij} (1-\widehat{\sigma}_{ij}) \Bigr\|_{\ell_2}
\qquad \text{(need } \geq I \text{ targets, or } \alpha_i = \alpha \text{)}
\end{align*}
\item \textbf{Price-cost margins}: Bertrand FOC with ownership matrix $\symbf{H}$ and demand derivatives $\Delta(\alpha) = \sum_i \widehat{w}_i \Delta_i(\alpha_i)$,
\begin{align*}
\symbf{c}(\alpha) = \symbf{p} - \bigl(\symbf{H} \odot \Delta(\alpha)\bigr)^{-1} \symbf{s},
\qquad
\min_{\alpha > 0}\ \bigl\| \mathcal{C}_j - c_j(\alpha) \bigr\|_{\ell_2}.
\end{align*}
\item Either way you now have a full mixed logit in product space: elasticities, diversion under price changes, merger simulation (Week 7).
\item \textbf{Inference}: each row of $\calD$ is $\text{Multinomial}(n_j, \cdot)$; parametric bootstrap with warm-started re-solves. Type labels are not identified, so report CIs for \alert{invariant functionals} only (fitted $D_{j\rightarrow k}$, elasticities). E.g.\ Suburban $\rightarrow$ Yukon XL $+$ Expedition: $65\%$, 95\% CI $[60\%, 71\%]$.
\end{wideitemize}
\end{frame}
